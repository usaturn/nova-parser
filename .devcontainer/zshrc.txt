function select_devcontainer_workspace() {
    local workspaces_root="${DEVCONTAINER_WORKSPACES_ROOT:-/workspaces}"
    local -a candidates
    local selected

    candidates=("${workspaces_root}"/*(N-/))
    case "${#candidates[@]}" in
        0)
            print -u2 "warning: no workspace directories found under ${workspaces_root}; using current directory"
            return 1
            ;;
        1)
            print -r -- "${candidates[1]}"
            return 0
            ;;
    esac

    if ! command -v fzf >/dev/null 2>&1; then
        print -u2 "warning: multiple workspace directories found but fzf not found; using current directory"
        return 1
    fi

    selected="$(printf '%s\n' "${candidates[@]}" | fzf)" || {
        print -u2 "warning: workspace selection cancelled; using current directory"
        return 1
    }
    if [ -z "$selected" ]; then
        print -u2 "warning: workspace selection cancelled; using current directory"
        return 1
    fi
    print -r -- "$selected"
}

# 素の `herdr` は「起動 or 既存セッションへ attach」。attach では選択した cwd が
# ワークスペースに反映されないため、サーバ稼働中は選択自体をスキップする。
# stale socket に騙されないよう、ファイルの有無ではなく CLI の出力で判定する。
function herdr_server_running() {
    command -v herdr >/dev/null 2>&1 || return 1
    herdr status server 2>/dev/null | grep -q '^status: running'
}

function herdrstart(){
    if [ "${HERDR_ENV:-}" = "1" ] || [ -n "${TMUX:-}" ] || [ ! -t 0 ] || [ ! -t 1 ]; then
        return 0
    fi
    export TZ="${TZ:-Asia/Tokyo}"
    if ! command -v herdr >/dev/null 2>&1; then
        print -u2 "warning: herdr not found; continuing without a multiplexer"
        return 0
    fi
    if command -v herdr-status-updater >/dev/null 2>&1; then
        herdr-status-updater --daemon || true
    fi
    local project_dir=""
    if ! herdr_server_running; then
        project_dir="$(select_devcontainer_workspace)" || project_dir=""
    fi
    if [ -n "$project_dir" ]; then
        (
            if builtin cd -- "$project_dir"; then
                herdr
            else
                print -u2 "warning: cannot enter ${project_dir}; using current directory"
                herdr
            fi
        )
    else
        herdr
    fi
}

function tmuxstart(){
    if [ -n "${TMUX:-}" ] || [ "${HERDR_ENV:-}" = "1" ]; then
        return 0
    fi
    local project_dir
    project_dir="$(select_devcontainer_workspace)" || project_dir="$PWD"
    export TZ="${TZ:-Asia/Tokyo}"
    tmux -u new-session -A -s yamada -n yamada -c "${project_dir}" \; \
        set-environment -g TZ "${TZ}"
}

# tmux ステータスバーの即時更新は Starship 管轄外のため precmd に残す。
function precmd_tmux_refresh() {
    if [ -n "${TMUX:-}" ]; then
        tmux refresh-client -S 2>/dev/null
    fi
}
precmd_functions+=(precmd_tmux_refresh)

# herdr 内で status updater が死んでいても、プロンプトのたびに起こす（flock で多重起動しない）
function precmd_herdr_status_updater() {
    if [ "${HERDR_ENV:-}" != "1" ]; then
        return 0
    fi
    if ! command -v herdr-status-updater >/dev/null 2>&1; then
        return 0
    fi
    herdr-status-updater --daemon >/dev/null 2>&1 || true
}
precmd_functions+=(precmd_herdr_status_updater)
